# === COMPILAR ===
arduino-cli compile \
  --fqbn esp32:esp32:esp32:PartitionScheme=huge_app \
  --libraries pos_kix_capacitivo_tamc_gt911/libraries \
  --jobs 4 \
  --build-property compiler.c.extra_flags="-Wno-error=implicit-function-declaration -Wno-error=implicit-int -Wno-implicit-function-declaration" \
  --verbose \
  pos_kix_capacitivo_tamc_gt911

# === UPLOAD (GRAVAR NA PLACA) ===
arduino-cli upload \
  -p /dev/ttyUSB0 \
  --fqbn esp32:esp32:esp32:PartitionScheme=huge_app \
  --verbose \
  pos_kix_capacitivo_tamc_gt911

# === COMPILAR E GRAVAR EM UMA SÓ LINHA ===
arduino-cli compile \
  --fqbn esp32:esp32:esp32:PartitionScheme=huge_app \
  --libraries pos_kix_capacitivo_tamc_gt911/libraries \
  --jobs 4 \
  --build-property compiler.c.extra_flags="-Wno-error=implicit-function-declaration -Wno-error=implicit-int -Wno-implicit-function-declaration" \
  --verbose \
  pos_kix_capacitivo_tamc_gt911 \
&& \
arduino-cli upload \
  -p /dev/ttyUSB0 \
  --fqbn esp32:esp32:esp32:PartitionScheme=huge_app \
  --verbose \
  pos_kix_capacitivo_tamc_gt911

# === MONITOR SERIAL ===
arduino-cli monitor -p /dev/ttyUSB0 -c baudrate=115200

# === EXPORTAR BINÁRIO ===
arduino-cli compile \
  --verbose \
  --fqbn esp32:esp32:esp32:PartitionScheme=huge_app \
  --libraries pos_kix_capacitivo_tamc_gt911/libraries \
  --jobs 4 \
  --build-property compiler.c.extra_flags="-Wno-error=implicit-function-declaration -Wno-error=implicit-int -Wno-implicit-function-declaration" \
  --output-dir ./build_output \
  pos_kix_capacitivo_tamc_gt911
